router = APIRouter(prefix="/projects", tags=["gaps"])


def apply_labels(claims, report):
    # Reset everything, then mark the problem claims
    for c in claims:
        c.label = "VERIFIED"
    for c in report["stale_tasks"]:
        c.label = "UNCERTAIN"
    for c in report["conflicting_deadlines"]:
        c.label = "CONFLICTING"
    for c in report["unowned_tasks"] + report["broken_dependencies"]:
        c.label = "MISSING"


@router.get("/{project_id}/gaps")

        "readiness_score": report["readiness_score"],
        "unowned_tasks": [claim_to_dict(c) for c in report["unowned_tasks"]],
        "conflicting_deadlines": [claim_to_dict(c) for c in report["conflicting_deadlines"]],
        "stale_tasks": [claim_to_dict(c) for c in report["stale_tasks"]],

TASK 2.1 - Validate every claim with Pydantic
---------------------------------------------
The proposal says you use Pydantic schema validation. This file
defines what a valid claim looks like. Invalid items are rejected.

##### FILE: backend/app/services/schemas.py ##### (NEW FILE)

from datetime import date
from typing import Literal, Optional
from pydantic import BaseModel, field_validator


class ClaimItem(BaseModel):
    kind: Literal["task", "deadline", "decision"]
    subject: str
    owner: Optional[str] = None
    due_date: Optional[str] = None
    status: Optional[str] = None
    depends_on: Optional[str] = None
    last_updated: Optional[str] = None
    quote: str

    @field_validator("due_date", "last_updated")
    @classmethod
    def check_date_format(cls, value):
        if value is None:
            return value
        date.fromisoformat(value)  # raises an error if not YYYY-MM-DD
        return value

    @field_validator("quote")
    @classmethod
    def quote_not_empty(cls, value):
        if not value.strip():
            raise ValueError("quote must not be empty")
        return value

The prompt tells the LLM to return only JSON. Keep it short and

##### FILE: backend/app/services/extractor.py ##### (REPLACE ENTIRE FILE)

import json
import os
from openai import OpenAI
from pydantic import ValidationError
from app.services.schemas import ClaimItem

PROMPT = """You extract project handover facts from text.
Return ONLY a JSON list. No explanation and no markdown.
Each item must have these keys:
kind: one of task, deadline, decision
subject: task name or milestone name
owner: person name, or null if not written in the text
due_date: YYYY-MM-DD, or null
status: To Do, In Progress, Done, or null
depends_on: name of the prerequisite task, or null
last_updated: YYYY-MM-DD, or null
quote: copy the exact sentence or line from the text
Only include facts that are written in the text. Do not guess.
If the same fact appears twice, include it twice with its own quote."""


def _clean_json(raw: str) -> str:
    raw = raw.strip()
    raw = raw.removeprefix("```json").removeprefix("```")
    raw = raw.removesuffix("```")
    return raw.strip()


def validate_items(raw_items):
    """Keeps valid claims and counts the invalid ones."""
    valid, invalid = [], 0
    for item in raw_items:
        try:
            valid.append(ClaimItem(**item).model_dump())
        except (ValidationError, TypeError):
            invalid += 1
    return valid, invalid


def extract_claims(text: str) -> list:
    api_key = os.getenv("OPENAI_API_KEY")
    if not api_key or api_key == "your_key_here":
        raise RuntimeError("OPENAI_API_KEY is not set in backend/.env")

    client = OpenAI(api_key=api_key)
    response = client.chat.completions.create(
        model=os.getenv("LLM_MODEL", "gpt-4o-mini"),
        messages=[
            {"role": "system", "content": PROMPT},
            {"role": "user", "content": text},
        ],
        temperature=0,
    )

    raw = response.choices[0].message.content
    items = json.loads(_clean_json(raw))
    if not isinstance(items, list):
        raise ValueError("LLM did not return a list")

    valid, invalid = validate_items(items)
    print(f"Extraction: {len(valid)} valid, {invalid} invalid claims")
    return valid

TASK 2.4 - Benchmark evaluation
-------------------------------
This script runs the real pipeline on all four sample files and
compares the flagged results with the planted errors (the answer
key). It reports recall (how many planted errors were found) and
precision (how many flagged items were correct).

Run it AFTER the OpenAI key is set. Results can change slightly
between runs, so run it three times and report the average.

##### FILE: backend/run_benchmark.py ##### (NEW FILE)

from types import SimpleNamespace as NS
from app.services.extractor import extract_claims
from app.services.gaps import run_all_checks

SAMPLE_FILES = [
    "../sample_data/meeting_notes.md",
    "../sample_data/task_board.csv",
    "../sample_data/README.md",
    "../sample_data/chat_log.txt",
]

# The answer key: what the sample data should be flagged for
EXPECTED = {
    "unowned": ["SMS notification service", "Export to CSV", "Email reminders"],
    "conflict": ["deployment", "report"],
    "stale": ["SMS notification service"],
    "broken": ["Admin dashboard"],
}

DEFAULTS = dict(kind=None, subject=None, owner=None, due_date=None,
                status=None, depends_on=None, last_updated=None)


def load_text():
    parts = []
    for path in SAMPLE_FILES:
        with open(path, encoding="utf-8") as f:
            parts.append(f"--- {path} ---\n" + f.read())
    return "\n\n".join(parts)


def matches(found, expected):
    if not found:
        return False
    f, e = found.lower(), expected.lower()
    return e in f or f in e


def score(flagged, expected):
    found = [e for e in expected if any(matches(c.subject, e) for c in flagged)]
    correct = [c for c in flagged if any(matches(c.subject, e) for e in expected)]
    recall = len(found) / len(expected) if expected else 1.0
    precision = len(correct) / len(flagged) if flagged else 0.0
    return recall, precision


def main():
    raw_items = extract_claims(load_text())
    claims = [NS(**{**DEFAULTS, **item}) for item in raw_items]
    print(f"Total claims extracted: {len(claims)}")

    report = run_all_checks(claims)
    checks = [
        ("Unowned tasks", report["unowned_tasks"], EXPECTED["unowned"]),
        ("Conflicting deadlines", report["conflicting_deadlines"], EXPECTED["conflict"]),
        ("Stale tasks", report["stale_tasks"], EXPECTED["stale"]),
        ("Broken dependencies", report["broken_dependencies"], EXPECTED["broken"]),
    ]

    print("\nCategory              Recall   Precision   Flagged")
    for name, flagged, expected in checks:
        recall, precision = score(flagged, expected)
        print(f"{name:<22}{recall:>6.0%}{precision:>11.0%}{len(flagged):>10}")


if __name__ == "__main__":
    main()

const API_URL = "http://127.0.0.1:8000";

async function request(path, options = {}) {
  const response = await fetch(`${API_URL}${path}`, options);
  const data = await response.json();
  if (!response.ok) {
    throw new Error(data.detail || "Request failed");
  }
  return data;
}

import LabelBadge from "./LabelBadge";

export default function Timeline({ claims }) {
  const dated = claims
    .filter((c) => c.due_date)
    .sort((a, b) => a.due_date.localeCompare(b.due_date));
  const undated = claims.filter((c) => !c.due_date);

  return (
    <div>
      <h2>Timeline</h2>
      {dated.length === 0 && <p>No dated items yet.</p>}
      <ul>
        {dated.map((c) => (
          <li key={c.id}>
            <strong>{c.due_date}</strong> - {c.subject}
            <LabelBadge label={c.label} />
          </li>
        ))}
      </ul>

      <h3>Undated items</h3>
      <ul>
        {undated.map((c) => (
          <li key={c.id}>
            {c.subject}
            <LabelBadge label={c.label} />
          </li>
        ))}
      </ul>
    </div>
  );
}


TASK 3.4 - Split-screen source viewer with highlight

Left: claim list. Right: the source text with the claim highlighted.
The text is split into three parts and only the middle part is

##### FILE: frontend/src/components/SourceViewer.jsx ##### (NEW FILE)

import { useEffect, useState } from "react";
import { getSource } from "../api";
import LabelBadge from "./LabelBadge";

export default function SourceViewer({ projectId, claims }) {
  const [selected, setSelected] = useState(null);
  const [source, setSource] = useState(null);
  const [error, setError] = useState("");

  useEffect(() => {
    if (!selected) return;
    setError("");
    setSource(null);
    getSource(projectId, selected.id)
      .then(setSource)
      .catch((e) => setError(e.message));
  }, [selected, projectId]);

  return (
    <div style={{ display: "flex", gap: "1rem" }}>
      <div style={{ width: "35%", maxHeight: "75vh", overflowY: "auto" }}>
        {claims.map((c) => (
          <div
            key={c.id}
            onClick={() => setSelected(c)}
            style={{
              cursor: "pointer",
              padding: 8,
              border: "1px solid #ccc",
              marginBottom: 6,
              background: selected && selected.id === c.id ? "#eef3ff" : "white",
            }}
          >
            <strong>{c.subject}</strong>
            <LabelBadge label={c.label} />
            <div style={{ fontSize: 12, marginTop: 4 }}>{c.quote}</div>
          </div>
        ))}
      </div>

      <div style={{ flex: 1 }}>
        {error && <p style={{ color: "red" }}>{error}</p>}
        {!selected && <p>Click a claim to see its source text.</p>}
        {selected && !source && !error && <p>Loading source...</p>}
        {source && <HighlightedText source={source} />}
      </div>
    </div>
  );
}

function HighlightedText({ source }) {
  const before = source.text.slice(0, source.start_char);
  const middle = source.text.slice(source.start_char, source.end_char);
  const after = source.text.slice(source.end_char);

  return (
    <div>
      <p>
        <em>{source.filename}</em>
      </p>
      <pre style={{ whiteSpace: "pre-wrap", fontFamily: "inherit" }}>
        {before}
        <mark>{middle}</mark>
        {after}
      </pre>
    </div>
  );
}

export default function ExportButton({ claims, projectId }) {
  function download() {
    const lines = ["# ContinuityOS Handover", "", `Project: ${projectId}`, ""];
    for (const c of claims) {
      const owner = c.owner ? `owner: ${c.owner}` : "no owner";
      const due = c.due_date ? ` - due ${c.due_date}` : "";
      lines.push(`- [${c.label}] ${c.subject} (${owner})${due}`);
      lines.push(`  Source quote: "${c.quote}"`);
    }

    const blob = new Blob([lines.join("\n")], { type: "text/markdown" });
    const url = URL.createObjectURL(blob);
    const link = document.createElement("a");
    link.href = url;
    link.download = `handover_project_${projectId}.md`;
    link.click();
    URL.revokeObjectURL(url);
  }

  return <button onClick={download}>Export Markdown</button>;
}

import { useState } from "react";
import { getClaims, uploadFile } from "./api";
import Timeline from "./components/Timeline";
import SourceViewer from "./components/SourceViewer";
import GapPanel from "./components/GapPanel";
import ExportButton from "./components/ExportButton";

const TABS = [
  { key: "timeline", label: "Timeline" },
  { key: "source", label: "Source viewer" },
  { key: "gaps", label: "Gaps" },
];

export default function App() {
  const [projectId, setProjectId] = useState("1");
  const [claims, setClaims] = useState([]);
  const [tab, setTab] = useState("timeline");
  const [message, setMessage] = useState("");

  async function loadClaims() {
    try {
      const data = await getClaims(projectId);
      setClaims(data);
      setMessage(`Loaded ${data.length} claims.`);
    } catch (e) {
      setMessage(e.message);
    }
  }

  async function handleUpload(event) {
    const file = event.target.files[0];
    if (!file) return;
    try {
      const result = await uploadFile(projectId, file);
      setMessage(`Saved ${result.claims_saved} claims, rejected ${result.claims_rejected}.`);
      await loadClaims();
    } catch (e) {
      setMessage(e.message);
    }
  }

  return (
    <div style={{ padding: "1.5rem", fontFamily: "Arial, sans-serif" }}>
      <h1>ContinuityOS</h1>

      <div style={{ marginBottom: 12 }}>
        Project ID:{" "}
        <input
          value={projectId}
          onChange={(e) => setProjectId(e.target.value)}
          style={{ width: 60 }}
        />{" "}
        <button onClick={loadClaims}>Load claims</button>{" "}
        <input type="file" onChange={handleUpload} />
      </div>

      <p>{message}</p>

      <div style={{ marginBottom: 12 }}>
        {TABS.map((t) => (
          <button
            key={t.key}
            onClick={() => setTab(t.key)}
            style={{ fontWeight: tab === t.key ? "bold" : "normal", marginRight: 6 }}
          >
            {t.label}
          </button>
        ))}
        <ExportButton claims={claims} projectId={projectId} />
      </div>

      {tab === "timeline" && <Timeline claims={claims} />}
      {tab === "source" && <SourceViewer projectId={projectId} claims={claims} />}
      {tab === "gaps" && <GapPanel projectId={projectId} />}
    </div>
  );
}
